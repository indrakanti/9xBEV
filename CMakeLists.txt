cmake_minimum_required(VERSION 3.18)
project(9xBEV LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(BEV_APP "Build YAML-configured executable" ON)
option(BEV_OPENGL "Build portable OpenGL ES 3 / EGL backend" OFF)
option(BEV_CUDA "Build NVIDIA CUDA backend" OFF)
include(CTest)
add_library(bev_core src/core.cpp src/image.cpp)
target_include_directories(bev_core PUBLIC include)
target_compile_options(bev_core PRIVATE -Wall -Wextra -Wpedantic)
if(BEV_OPENGL)
 find_path(EGL_INCLUDE_DIR EGL/egl.h REQUIRED)
 find_path(GLES_INCLUDE_DIR GLES3/gl3.h REQUIRED)
 find_library(EGL_LIBRARY EGL REQUIRED)
 find_library(GLES_LIBRARY GLESv2 REQUIRED)
 add_library(bev_opengl src/opengl.cpp)
 target_include_directories(bev_opengl PRIVATE ${EGL_INCLUDE_DIR} ${GLES_INCLUDE_DIR})
 target_link_libraries(bev_opengl PUBLIC bev_core ${EGL_LIBRARY} ${GLES_LIBRARY})
endif()
if(BEV_CUDA)
 enable_language(CUDA)
 add_library(bev_cuda src/cuda.cu)
 set_target_properties(bev_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
 target_link_libraries(bev_cuda PUBLIC bev_core)
endif()
if(BEV_APP)
 find_package(yaml-cpp REQUIRED)
 add_library(bev_config src/config.cpp)
 if(TARGET yaml-cpp::yaml-cpp)
  target_link_libraries(bev_config PUBLIC bev_core yaml-cpp::yaml-cpp)
 else()
  target_link_libraries(bev_config PUBLIC bev_core yaml-cpp)
 endif()
 add_executable(bev_render src/main.cpp)
 target_link_libraries(bev_render PRIVATE bev_config)
 if(BEV_OPENGL)
  target_link_libraries(bev_render PRIVATE bev_opengl)
  target_compile_definitions(bev_render PRIVATE BEV_HAS_OPENGL)
 endif()
 if(BEV_CUDA)
  target_link_libraries(bev_render PRIVATE bev_cuda)
  target_compile_definitions(bev_render PRIVATE BEV_HAS_CUDA)
 endif()
endif()
if(BUILD_TESTING)
 add_executable(bev_tests tests/core_tests.cpp)
 target_link_libraries(bev_tests PRIVATE bev_core)
 add_test(NAME core COMMAND bev_tests)
 if(BEV_APP)
  add_executable(config_tests tests/config_tests.cpp)
  target_link_libraries(config_tests PRIVATE bev_config)
  add_test(NAME config COMMAND config_tests ${CMAKE_CURRENT_SOURCE_DIR}/config/qualcomm_linux.yaml)
 endif()
 if(BEV_OPENGL)
  add_executable(gpu_tests tests/gpu_tests.cpp)
  target_link_libraries(gpu_tests PRIVATE bev_opengl)
  add_test(NAME opengl_parity COMMAND gpu_tests)
 endif()
 if(BEV_CUDA)
  add_executable(cuda_tests tests/gpu_tests.cpp)
  target_compile_definitions(cuda_tests PRIVATE TEST_CUDA)
  target_link_libraries(cuda_tests PRIVATE bev_cuda)
  add_test(NAME cuda_parity COMMAND cuda_tests)
 endif()
endif()
